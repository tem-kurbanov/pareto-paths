cmake_minimum_required(VERSION 3.20)
project(pareto_paths VERSION 0.1.0 LANGUAGES CXX)
option(PARETO_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(PARETO_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(pareto src/graph.cpp src/cover.cpp src/search.cpp src/io.cpp)
add_library(pareto::pareto ALIAS pareto)
target_include_directories(pareto PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
add_executable(pareto-paths src/main.cpp)
target_link_libraries(pareto-paths PRIVATE pareto)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(pareto PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(pareto-paths PRIVATE -Wall -Wextra -Wpedantic)
  if(PARETO_WARNINGS_AS_ERRORS)
    target_compile_options(pareto PRIVATE -Werror)
    target_compile_options(pareto-paths PRIVATE -Werror)
  endif()
  if(PARETO_SANITIZERS)
    target_compile_options(pareto PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(pareto PUBLIC -fsanitize=address,undefined)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(pareto-tests tests/test_main.cpp)
  target_link_libraries(pareto-tests PRIVATE pareto)
  add_test(NAME algorithms COMMAND pareto-tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:pareto-paths> ${CMAKE_CURRENT_SOURCE_DIR})
  endif()
endif()
